// ── FILTER ────────────────────────────────────────
function activeTypes() {
return TYPES.filter(t => {
for (const [dim, val] of Object.entries(choices)) {
if (!val) continue;
if (t.pol[dim] && t.pol[dim] !== val) return false;
}
if (centrumFilter && t.centrum !== centrumFilter) return false;
return true;
});
}

// ── PEEK: highlight polariteiten van aangeklikt type ──
function applyPeek() {
// Clear all peek states first
document.querySelectorAll('.pol-btn.peeked').forEach(b => b.classList.remove('peeked'));

const legend = document.getElementById('peek-legend');

if (!peekType) {
legend.classList.remove('visible');
return;
}

const t = TYPES.find(x => x.nr === peekType);
if (!t) return;

// For each dimension, highlight the button matching this type's comfortable side
DIMS.forEach(d => {
const comfortVal = t.pol[d.id];
if (!comfortVal) return;
const btn = document.getElementById(`pb-${d.id}-${comfortVal}`);
if (btn && !btn.classList.contains('selected')) {
btn.classList.add('peeked');
}
});

legend.classList.add('visible');
document.getElementById('peek-legend-text').textContent =
`Blauw = polariteiten van Type ${t.nr} · ${t.naam} (nog niet gekozen)`;
}

// ── RENDER ────────────────────────────────────────
function render() {
const active = activeTypes();
const anrs = active.map(t => t.nr);

// Sidebar
const sc = document.getElementById('type-cards');
sc.innerHTML = '';
TYPES.forEach(t => {
const on = anrs.includes(t.nr);
const isPeeked = peekType === t.nr;
const d = document.createElement('div');
let cls = 'type-card';
if (!on) cls += ' eliminated';
else if (isPeeked) cls += ' peeked';
d.className = cls;
d.onclick = () => {
if (!on) return;
peekType = isPeeked ? null : t.nr; // toggle
render();
applyPeek();
};
d.innerHTML = `
<div class="tc-top">
<div class="type-badge badge-${t.centrum}">${t.nr}</div>
<div class="tc-name">Type ${t.nr} · ${t.naam}</div>
</div>
<div class="tc-sub">${t.verleiding} · ${t.centrum}</div>
<div class="elim-tag">geëlimineerd</div>`;
sc.appendChild(d);
});

document.getElementById('active-count').textContent = `${active.length} van 9 types actief`;

// Result banner
const ban = document.getElementById('result-banner');
if (active.length === 1) {
const t = active[0];
ban.classList.add('visible');
document.getElementById('r-title').textContent = `Type ${t.nr} · ${t.naam}`;
document.getElementById('r-meta').textContent = `${t.centrum}type · ${t.pol['Match/Mismatch']}`;
document.getElementById('r-credo').textContent = t.credo;
document.getElementById('r-chips').innerHTML =
[['Verleiding',t.verleiding],['Vermijding',t.vermijding],['Verdediging',t.verdediging],['Verslaving',t.verslaving]]
.map(([k,v]) => `<div class="r-chip"><span class="chip-k">${k}</span><span class="chip-v">${v}</span></div>`)
.join('');
} else {
ban.classList.remove('visible');
}

renderVC(active);
renderKK(active);
}

// ── INTERACTIONS ──────────────────────────────────
function selPol(dim, val) {
// Selecting a polariteit clears peek mode
peekType = null;
document.getElementById('peek-legend').classList.remove('visible');

choices[dim] = choices[dim] === val ? undefined : val;

DIMS.find(d => d.id === dim).opts.forEach(o => {
const btn = document.getElementById(`pb-${dim}-${o}`);
if (btn) {
btn.classList.remove('peeked');
btn.classList.toggle('selected', choices[dim] === o);
}
});

const bar = document.getElementById('pol-effect');
if (choices[dim]) {
const elim = TYPES.filter(t => t.pol[dim] && t.pol[dim] !== choices[dim]);
bar.className = 'effect-bar has';
bar.textContent = elim.length
? `"${choices[dim]}" elimineert: ${elim.map(t => `Type ${t.nr} (${t.naam})`).join(', ')}`
: `"${choices[dim]}" — geen eliminaties op deze dimensie.`;
} else {
bar.className = 'effect-bar';
bar.textContent = 'Selecteer een polariteit om te zien welke types worden geëlimineerd.';
}
render();
}

function filterCentrum(c) {
centrumFilter = c;
['Hart','Hoofd','Buik','Alle'].forEach(id => {
const b = document.getElementById('btn-' + id);
if (b) b.className = 'centrum-btn';
});
const id = c ? 'btn-' + c : 'btn-Alle';
const cls = c ? `centrum-btn sel-${c.toLowerCase()}` : 'centrum-btn sel-all';
const b = document.getElementById(id);
if (b) b.className = cls;
render();
}

function showTab(n) {
[1,2,3,4,5,6].forEach(i => {
const el = document.getElementById('tab-' + i);
if (el) el.style.display = i === n ? '' : 'none';
});
document.querySelectorAll('.tab').forEach((t, i) => t.classList.toggle('active', i === n - 1));
render();
}

function resetAll() {
choices = {}; centrumFilter = null; peekType = null;
document.querySelectorAll('.pol-btn').forEach(b => { b.classList.remove('selected'); b.classList.remove('peeked'); });
document.getElementById('peek-legend').classList.remove('visible');
const bar = document.getElementById('pol-effect');
bar.className = 'effect-bar';
bar.textContent = 'Selecteer een polariteit om te zien welke types worden geëlimineerd.';
['Hart','Hoofd','Buik'].forEach(id => { const b = document.getElementById('btn-' + id); if (b) b.className = 'centrum-btn'; });
const alle = document.getElementById('btn-Alle');
if (alle) alle.className = 'centrum-btn sel-all';
render();
}
